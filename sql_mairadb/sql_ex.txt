-- DB의 무결성
-- 종류 - 개체 무결성 / 참조 무결성 / 키 무결성 / 도메인 무결성



-- < 1. 개체 무결성 > (primary key 제약조건)
-- 기본키(primary key, pk) 제약 조건 : 기본 키는 빈 값(Null)이나 중복 값을 가질 수 없다(= Not Null). 자동으로 인덱스가 생성된다.


-- 참고 : 테이블 작성시 칼럼의 이름, 타입은 중요하지만 순서는 마음대로 해도 된다. 
--        계산에 의해 처리될 수 있는 값은 칼럼으로 작성하지 않는다.
--        ex) 국어, 영어따위는 칼람으로 작성하지만 총점 계산이 불가능하다. **절대로 총점과 평균은 칼럼으로 작성하지 않는다**

-- 방법1. 칼럼 레벨
CREATE TABLE aa(bun INT PRIMARY KEY, irum CHAR(10)); -- 테이블을 생성함
DESCRIBE aa; -- aa테이블을 보여줘.
INSERT INTO aa VALUES(1, 'tom');
INSERT INTO aa VALUES(2, 'tom');
-- INSERT INTO aa VALUES(2, 'tom'); -- pk error ... 동일 데이터가 중복되었기 때문
-- INSERT INTO aa(irum) VALUES('tom'); -- pk error ... Not Null인데 Null에 값을 줌
-- Primary key는 웬만하면 수정하지 않는다.
SELECT * FROM aa;


-- 제약 조건 확인
SELECT * FROM information_schema.TABLE_CONSTRAINTS WHERE TABLE_NAME='aa'; 

DROP TABLE aa;


-- 방법2. 테이블 레벨
CREATE TABLE aa(bun INT, irum CHAR(10), CONSTRAINT aa_bun_pk PRIMARY KEY(bun)); -- 칼럼을 주어주고 CONSTRAINT가 주어진 상황
DESCRIBE aa;
SELECT * FROM information_schema.TABLE_CONSTRAINTS WHERE TABLE_NAME='aa';

-- alter TABLE aa DROP CONSTRAINT aa_bun_pk; -- pk 제약 조건 삭제 ... 오라클에서만 유효
-- alter TABLE aa DROP PRIMARY KEY -- pk 제약 조건 삭제 이런식으로 해도 된다.
DROP TABLE aa;

-- 일반적으로 방법2보다 방법1을 더 많이 쓴다.



-- < 2. 도메인 무결성 > (check 제약조건) : 입력 값에 조건 부여
CREATE TABLE aa(bun INT, irum CHAR(10), nai INT CHECK(nai >=20)); -- nai에 대해서 조건을 주었다.
SELECT * FROM information_schema.TABLE_CONSTRAINTS WHERE TABLE_NAME='aa';

-- < 3. 키 무결성 > (unique 제약조건) : 입력값에 조건 부여(동입값 입력 불허)
CREATE TABLE aa(bun INT, irum CHAR(10) UNIQUE); 
INSERT INTO aa VALUES(1, 'john');
INSERT INTO aa VALUES(2, 'tom');
-- INSERT INTO aa VALUES(3, 'tom'); -- unique error : 중복불가
-- 테이블을 만들면서 컬럼의 순서는 중요하지 않다.
DROP TABLE aa;



-- < 4. 참조 무결성 > (참조키(외래키(foreign key)) 제약조건) : 다른 테이블의 칼럼값을 참조
-- 다른 테이블의 칼럼값을 참조(foreign key의 대상은 다른 테이블의 pk 또는 unique 가능)

-- 기존 테이블에 새로운 칼럼을 추가할 때 그 칼럼의 정보가 개인정보라면 따로 별개의 테이블을 만든다. 
-- 그 때 두 테이블간의 연결고리가 있어야 join이 된다. 
-- 그때 중복데이터가 가능한 컬럼은 쓰지 않는다. 그때 새로 생생성된 칼럼이 기존 칼럼을 참조하고 있을 때 
-- foreign key가 primary key를 참조한다.

-- 직원 table(부모table)
CREATE TABLE jikwon(bun INT PRIMARY KEY, irum VARCHAR(10) NOT NULL, buser CHAR(10));

-- 가족 table(자식table)
CREATE TABLE gajok(CODE INT PRIMARY KEY, NAME VARCHAR(10), birth DATETIME, jikwon_bun INT, 
FOREIGN KEY(jikwon_bun) REFERENCES jikwon(bun));

-- RESTRICT  VS  CASCADE
CREATE TABLE gajok(CODE INT PRIMARY KEY, NAME VARCHAR(10), birth DATETIME, jikwon_bun INT, 
FOREIGN KEY(jikwon_bun) REFERENCES jikwon(bun) ON DELETE RESTRICT ON UPDATE RESTRICT); -- 위와 동일1
-- 부모 데이터를 자식이 참조하고 있으면 부모 table을 삭제하거나 키 변경을 막는다. 자식 table을 남겨두고 부모 table의 컬럼을 제거할 순 없다. 
-- 그러나 그 때 CASCADE를 쓰면 부모것도 같이 삭제가 된다.

CREATE TABLE gajok(CODE INT PRIMARY KEY, NAME VARCHAR(10), birth DATETIME, jikwon_bun INT, 
FOREIGN KEY(jikwon_bun) REFERENCES jikwon(bun) ON DELETE CASCADE ON UPDATE SET NULL); -- 위와 동일2
-- CASCADE : 부모 table 삭제 시 자식도 삭제된다.
-- SET NULL : 부모 참조 key 변경 시 자식의 foreign key의 칼럼 값은 null이 된다.
-- 대게의 경우 부모의 primary key는 변경이 흔하지 않으므로 SET NULL은 자주 사용되지 않는다.



DESC gajok;

-- 인덱스(index) : 인덱스는 쉽게 말하면 책의 목차나 색인과 비슷하다.
-- 인덱스 없음 = 처음부터 끝까지 데이터를 확인
-- 인덱스 있음 = 위치를 빠르게 찾아서 데이터 접근
-- 목표 : 검색 속도 증진이 목표
-- B-tree : 절반씩 나눠가면서 값에 가까운 값을 찾아가는 방법, 이름과 ptr이 있는 색인표에서 인덱스를 할 때, 검색 속도가 빠르다.
-- 하지만 insert, update, delete를 쓰는 경우 색인이 계속 수정되어 색인을 다시 매겨야 하기 때문에 그럴 때는 좋지 않다.

-- 1. lower, upper 
SELECT LOWER('hello SQL'), UPPER('hello SQL') FROM DUAL; -- 대소문자로 변환한 함수
-- DUAL : 연산을 위해서 임의의 table을 만들어 둔 것이다.
-- lower : 소문자로 표기
-- upper : 대문자로 표기

-- 2. Concat 
SELECT CONCAT('Hello' 'World') FROM DUAL; 
-- concat('Hello' 'World')라는 하나의 열이 생기고 연산을 진행한다.

-- 3. Substr
SELECT SUBSTR('Hello World', 3), SUBSTR('Hello World', 3, 3), SUBSTR('Hello World', -3, 3); -- 문자열 일부 추출
-- SUBSTRING(문자열, 시작위치, 가져올 글자수)
-- 0번째 부터 ...
-- ('Hello World', 3) : 0부터 3개의값
-- ('Hello World', 3, 3) : 3부터 3개의 값
-- ('Hello World', -3, 3) : 뒤에서부터: -3부터 3개의 값
-- from dual은 있어도 되고 없어도 된다. 연산을 위해서 우리가 임의로 넣어준 것이다.

-- 4. length
SELECT LENGTH('Hello World') FROM DUAL;
-- : Hello World의 길이

-- 5. Instr
SELECT INSTR('Hello World', 'e'); 
-- 특정문자 위치를 반환
-- 'Hello World'에서 문자 e는 몇 번째에 나와?

SELECT INSTR('Hello World', 'k'); 
-- 내가 찾고자 하는 문자가 없으면 0이 나온다.

-- 6. Trim, Ltrim, Rtrim
SELECT TRIM(' aabb bbaa '); -- 앞뒤의 공백(둘 다) 제거
SELECT LTRIM(' aabb bbaa'); -- 가장 왼쪽의 공백만 제거
SELECT RTRIM(' aabb bbaa'); -- 가장 오른쪽의 공백만 제거

-- 7. Replace 문자열을 치환할 수 있다.

-- 조건 :마지막에 '이'가 있는 경우는 한 글자 출력
--      : 열에 대한 작업과 행에 대한 작업도 필요하다.
--      : 이름에 마지막에 '이'가 있는 경우 -> '이'만 출력

-- 문자열 중 중간에 이가 들어가 있는 애들은 문자열의 개수에 상관없이 모두 출력해라
SELECT jikwonname,
CASE-- 이름 처음이 '이'인 경우
           WHEN INSTR(jikwonname, '이') = 1
               THEN CONCAT('이', '-')

           -- 이름 마지막이 '이'인 경우
           WHEN INSTR(jikwonname, '이') = CHAR_LENGTH(jikwonname)
               THEN SUBSTR(jikwonname, INSTR(jikwonname, '이'))

           -- 이름 중간에 '이'가 있는 경우
           ELSE SUBSTR(jikwonname, INSTR(jikwonname, '이'), 2)
       END AS 결과
FROM jikwon
WHERE jikwonname LIKE '%이%';

 -- 1. Round
SELECT ROUND(12.5456), ROUND(12.5678, 0), ROUND(12.3456, 1), ROUND(12.3456, -1), ROUND(17.3456, -1);

-- --- 기타함수 ------- rank(), dense_rank(), nvl(), nvl2(), nullif()

-- < 기타함수 >
-- rank() over함수, dense_rank() over함수 : 순위 결정 함수
SELECT jikwonno, jikwonname, jikwonpay,
RANK()OVER(ORDER BY jikwonpay), 
-- jikwonpay에 대해서 오름차순으로 순위를 매긴다
-- 급여가 높은 사람이 1위 = 뒤쪽 순위에 존재(오름차순)
-- 동점이 존재하면 그 순위는 동점 데이터 수만큼 건너뛴다. 1224등

DENSE_RANK()OVER(ORDER BY jikwonpay) -- 
-- jikwonpay에 대해서 오름차순으로 순위를 매긴다
-- 급여가 높은 사람이 1위
-- 동점 데이터가 존재하더라도 순서대로 간다. 단 동점자는 동일 순위. 1223등

FROM jikwon 

WHERE jikwonpay IS NOT NULL; -- 급여가 null인 jikwon은 제외하고 순위를 매긴다.


SELECT jikwonno, jikwonname, jikwonpay,
rank() OVER(ORDER BY jikwonpay desc), 
dense_rank() OVER(ORDER BY jikwonpay desc)
FROM jikwon WHERE jikwonpay IS NOT NULL; -- 내림차순

-- **결과 행을 정렬하는 ORDER BY가 아니라, 
-- RANK 함수가 순위를 계산할 기준을 정하는 ORDER BY이다. 

-- 일반적으로 보면 select  ->  from  ->  where  ->  order by  -> having 순이 맞다

-- rank() vs dense_rank()
-- 동점자 순위 처리에 차이가 있다.
-- rank() : 동점처리를 한다.
-- dense_rank() : 동점 처리를 하지 않는다. 건너뛴다.

SELECT jikwonno, jikwonname, jikwonibsail,
rank() OVER(ORDER BY jikwonibsail), 
dense_rank() OVER(ORDER BY jikwonibsail) -- 입사일을 오름차순으로 배치하고 dense_rank를 이용하여 순위를 결정하라
FROM jikwon; -- 입사 년도별



-- nvl(value1, value2) : value1이 null이면 value2를 취한다.
UPDATE jikwon SET jikwonjik=NULL WHERE jikwonno = 5; 
-- jikwonno가 5인 행들 중에서 jikwonjik을 null로 바꾸어라

SELECT 
jikwonname, 
nvl(jikwonjik, '임시직'), 
nvl(jikwonpay, 0) 
FROM jikwon;
-- jikwonjik과 jikwonpay 둘 다 null인 값에 대해서 nvl을 통해 값을 부여할 수 있다.
-- = nvl은 null값을 다른 값으로 바꿔서 보여준다.
-- nvl(a,b) : a가 NULL이면 b, a가 null이 아니고 값이 있으면 a가 그대로 출력.
-- 그런데 위 예시는 우리가 a를 null로 만들어줬기에 '임시직'이라는 대체 표현이 보이게 된것이다.


-- nvl2(value1, value2, value3) : value1이 null이면 value3을 취하고, val1이 null이 아니면 value2를 취한다.
SELECT 
jikwonname,
NVL2(jikwonjik, '정규직', '임시직') AS jik, 
-- jikwonjik에 대해서 null인 부분은 임시직이라고 표현하고 null이아니고 다른 문자열 값이 있으면 그 표현 대신 정규직이라고 수정한다.
jikwongen,
NVL2(jikwonpay, jikwonpay, 0) AS pay
-- jikwonpay에 대해서 값이 null이면 0을 보여주고 null이 아닌 다른 값이 있으면 그 값 그대로 jikwonpay의 값을 보여줘라.
FROM jikwon;

-- < 조건 표현식(Conditional Expressions) >
-- 형식1) case 표현식 when 비교값1 then 결과값1 / when 비교값2 then 결과값2 ... [else 결과값n] end as 별명;
SELECT 
case 10 / 5 
when 5 
then '안녕' 
-- 10을 5로 나누는 상황(case)에서 5가 나올때는 '안녕'을 출력하고
when 2 
then '수고' 
ELSE '잘가' 
END AS 결과 
-- 2가 나올 때는 '수고'를 출력해라.
-- 둘 다 아니라면 '잘 가'를 출력하고
-- 열을 '결과'라는 이름과 함께 출력해라.

FROM DUAL;
-- case 10 / 5

SELECT 
jikwonname, 
jikwonpay,
case jikwonpay -- case를 쓰려면 앞에서 언급된 컬럼을 가져오는게 일반적이다.
when 3000 
then '연봉 3000'
when 3500 
then '연봉 3500'
ELSE '기타연봉'
END AS result
FROM jikwon;

-- case를 통해서 내가 추출할 컬럼에 조건을 줄 수 있다.

SELECT 
jikwonname, 
jikwonpay, 
jikwonjik,
case jikwonjik 
when '이사' 
then jikwonpay * 0.05
when '부장' 
then jikwonpay * 0.04
when '과장' 
then jikwonpay * 0.03
ELSE jikwonpay * 0.02
END AS donation
FROM jikwon;
-- jikwon table에서 jikwonname, jikwonpay, jikwonjik 그리고 donation 컬럼을 데려와라
-- donation컬럼은 내가 연산을 위해서 만든 임의의 컬럼이다.
-- donation컬럼은 jikwonjik(앞에서 언급한 컬럼 중 하나)에 대해서 값이'이사'라면 기존 jikwonpay에 0.05를 곱하고 
-- '부장'이라면 0.04를 곱하고, 
-- '과장'이라면 0.03을 곱하고, 
-- 이 3가지 직급 모두 아니라면 0.02를 곱한 값을 출력한다. 그리고 그 결괏값을 컬럼으로 만들어 donation이라고 출력한다

FROM jikwon;

-- 오늘의 핵심 : subquery --- 중요!
-- subquery : SQL문 안에 포함된 또 다른 select문의 사용
-- 일반적으로 안쪽 select문을 실행해서 그 결과를 구한 후 바깥쪽 SQL문에서 사용하는 방식
-- 복잡한 조건을 단게적으로 처리가능하다.

-- '사번 10번인 박치기 / 직원과 / 직급이 같은 / 모든 직원 출력'

-- subquery 사용X
-- 1. 박치기 직급부터 구하고
-- 2.직급을 행조건으로 주어주고 jikwon 테이블 조회함 
SELECT jikwonjik 
FROM jikwon 
WHERE jikwonno=10; -- 사번이 10번인 직급들('사원')이 모두 나옴.
SELECT * FROM jikwon WHERE jikwonjik = '사원'; -- '사원'인 데이터 모두 출력
-- 우선 이 형태로 jikwonjik = 
-- 결과를 위해 2번의 SQL문을 사용한 것이다.
-- DB server를 2번 갔다 옴

-- subquery 사용ㅇ
SELECT * 
FROM jikwon 
WHERE jikwonjik = (SELECT jikwonjik 
                   FROM jikwon							
						 WHERE jikwonno = 10);
-- = select 문 안에 select문이 존재한다.()안의 값을 사원이라고 구하고 다시 select함.
-- DB server를 한번만 갔다 옴 = 정보처리속도가 빠르다. ... subquery의 장점
-- 괄호가 먼저 수행된다.

SELECT * 
FROM jikwon 
WHERE jikwonjik = (SELECT jikwonjik 
                   FROM jikwon 
						 WHERE jikwonno = (SELECT jikwonno 
						                   FROM jikwon 
												 WHERE jikwonno = 10));

-- 트랜잭션(transaction)은 여러 SQL 작업을 하나의 작업 단위로 묶어 처리하는 것.
-- 단위별 데이터 처리를 의미한다.
-- 예를 들어 '계좌이체'처럼
-- - A 계좌 출금 
-- - B 계좌 입금
-- a 계좌에서 출금하고 b 계좌에 입금되는 하나의 작업
-- 그 작업단위를 트랜잭션이라고 한다.
-- 두 작업이 모두 성공해야 하나의 정상 처리된다. 중간에 실패라면 원래의 작업으로 복귀한다.
-- 트랜잭션은 여러 SQL문을 하나의 논리적인 작업 단위로 처리하며, 
-- 모두 성공하면 COMMIT, 문제가 생기면 ROLLBACK 한다.
-- INSERT, UPDATE, DELETE같은 DML문이 실행되면 트랜잭션이 시작되며,
-- 명시적 트랜잭션에서는 COMMIT 또는 ROLLBACK을 만나면 종료된다.

-- Maria DB는 묵시적으로 INSERT, UPDATE, DELETE하면 자동 commit된다.
-- 즉, client의 자료를 근거로 db-server의 자료를 갱신한다.
-- 현재설정 확인 : select @@autocommit; -- 결과가 1이면 자동 커밋 상태이다.

-- commit : 원본이 수정된다
-- rollback : 내가 지금까지 한 update, insert, delete가 취소된다.

-- deadlock(교착상태):

-- 해결책은 충돌한 트랜잭션 하나를 commit 또는 rollback하라는 소리이다.
-- 시간과 자원을 낭비함으로 주의가 필요. 트랜잭션 완료 확인이 필요하다.

UPDATE jiktab SET jikwonjik = '주임' WHERE jikwonno = 4;
SELECT * FROM jiktab WHERE jikwonno = 4;
COMMIT; -- deadlock 해제
SELECT * FROM jiktab WHERE jikwonno = 4;

-- 실습 5 : if사용
-- 고객번호 입력하면 고객정보를 조회할 수 있다.
-- 담당사원번호(담당직원번호)를 입력하면 해당 직원이 관리하는 고객을 출력한다
-- 둘 다 입력하지 않으면 모든 고객을 출력한다.
delimiter $$
CREATE OR REPLACE PROCEDURE sp_5(IN p_gogekno VARCHAR(20), IN p_damsano INT)
BEGIN 
  -- 특정 고객 조회
  if p_gogekno IS NOT NULL then
    SELECT * FROM gogek WHERE gogekno = p_gogekno;
  -- 해당 직원이 관리하는 고객 출력
  ELSEIF p_damsano is NOT NULL then
    SELECT * FROM gogek WHERE gogekdamsano = p_damsano;
  -- 모든 고객 출력
  else
    SELECT * FROM gogek;
  END if;
END $$
delimiter ;

CALL sp_5(5, NULL); -- 고객번호 5번 조회
CALL sp_5(NULL, 3); -- 담당직원번호 3번이 관리하는 고객 조회
CALL sp_5(NULL, NULL);

delimiter $$
CREATE OR REPLACE PROCEDURE sp_7(IN start_no INT, IN end_no INT)
BEGIN
  DECLARE current_no INT;
  SET current_no = START_no;
  
  while current_no <= end_no do
    SELECT * FROM gogek WHERE gogekno = current_no;
    SET current_no = current_no + 1;
  END while;
  
END $$

-- 실습8 : while 사용
-- jikwon 테이블을 이용해 입력한 직급의 직원수와 직원 목록출력
delimiter $$
CREATE OR REPLACE PROCEDURE sp_8(IN p_jik VARCHAR(20))
BEGIN
  DECLARE cnt INT DEFAULT 0;
  DECLARE i INT DEFAULT 0;
  
  -- 해당 직급의 직원 수 저장
  SELECT COUNT(*) INTO cnt FROM jikwon WHERE jikwonjik = p_jik;
  
  SELECT p_jik AS 직급, cnt AS 인원수;
  
  -- 직원 수 만큼 출력
  while i < cnt do
    SELECT jikwonno, jikwonname, jikwonjik, jikwonpay FROM jikwon
    WHERE jikwonjik = p_jik
	 ORDER BY jikwonno ASC LIMIT i, 1;
	 SET i = i + 1;
  END while;
END $$
delimiter ;

CALL sp_8('대리');
